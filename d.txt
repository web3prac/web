[35mpragma solidity ^0.8.24;[0m

[35mcontract Voting {[0m

[35m    address public owner;[0m
[35m    uint public teamA;[0m
[35m    uint public teamB;[0m
[35m    uint public teamC;[0m
[35m    mapping(address => bool) public voted;[0m

[35m    constructor() { owner = msg.sender; }[0m

[35m    function voteA() public {[0m
[35m        require(!voted[msg.sender], "Already voted");[0m
[35m        voted[msg.sender] = true; teamA++;[0m
[35m    }[0m
[35m    function voteB() public {[0m
[35m        require(!voted[msg.sender], "Already voted");[0m
[35m        voted[msg.sender] = true; teamB++;[0m
[35m    }[0m
[35m    function voteC() public {[0m
[35m        require(!voted[msg.sender], "Already voted");[0m
[35m        voted[msg.sender] = true; teamC++;[0m
[35m    }[0m
[35m    function totalVotes() public view returns (uint) {[0m
[35m        return teamA + teamB + teamC;[0m
[35m    }[0m
[35m    function resultOfVoting() public view returns (string memory) {[0m
[35m        if (teamA > teamB && teamA > teamC) { return "A is Winning"; }[0m
[35m        else if (teamB > teamA && teamB > teamC) { return "B is Winning"; }[0m
[35m        else if (teamC > teamA && teamC > teamB) { return "C is Winning"; }[0m
[35m        else { return "Tie"; }[0m
[35m    }[0m
[35m}[0m
